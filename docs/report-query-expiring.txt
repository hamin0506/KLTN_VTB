DECLARE @f_DivisionIds1 int = 1;
DECLARE @term_contains nvarchar(4000) = N'%'' OR 1=1 --%';
DECLARE @p int = 20;

SELECT [v0].[Department], [v0].[Id], [v0].[Name], [v0].[Document], [v0].[Code], [v0].[IssueDate], [v0].[ExpiryDate], [v0].[Days]
FROM (
    SELECT DISTINCT NULL AS [Department], [v].[employee_id] AS [Id], [v].[full_name] AS [Name], [v].[document_name] AS [Document], [v].[document_code] AS [Code], NULL AS [IssueDate], [v].[expiry_date] AS [ExpiryDate], [v].[expiry_days] AS [Days]
    FROM [dbo].[vw_document_quality] AS [v]
    WHERE [v].[employee_id] IN (
        SELECT [e].[employee_id]
        FROM [dbo].[employee] AS [e]
        WHERE EXISTS (
            SELECT 1
            FROM [dbo].[department] AS [d]
            WHERE [d].[department_id] = [e].[department_id] AND [d].[division_id] = @f_DivisionIds1)
    ) AND [v].[quality_status] = N'SAP_HET_HAN' AND ([v].[employee_id] LIKE @term_contains ESCAPE N'\' OR [v].[full_name] LIKE @term_contains ESCAPE N'\' OR [v].[document_name] LIKE @term_contains ESCAPE N'\')
) AS [v0]
ORDER BY [v0].[Id], [v0].[Name], [v0].[Code], [v0].[Document], [v0].[ExpiryDate], [v0].[Days]
OFFSET @p ROWS FETCH NEXT @p ROWS ONLY
